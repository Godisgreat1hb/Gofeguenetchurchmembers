Supabase Database & Security Guide

Complete SQL schema, RLS policies, and setup instructions for Gofe Guenut Church.

1. Supabase PostgreSQL Schema SQL

Execute this SQL script inside your Supabase SQL Editor to create all required tables with proper relations and constraints:

-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- Table 1: members
create table public.members (
    id uuid default uuid_generate_v4() primary key,
    membership_number text unique not null,
    full_name text not null,
    date_of_birth date not null,
    gender text,
    phone text not null,
    email text,
    address text not null,
    kebele text not null,
    profile_image_url text,
    marital_status text,
    emergency_contact_name text not null,
    emergency_contact_phone text not null,
    relationship text not null,
    membership_category text not null,
    membership_status text default 'Active',
    joining_date date not null,
    baptism_status text,
    previous_church text,
    ministry text not null,
    education text,
    occupation text,
    skills text,
    additional_notes text,
    consent_given boolean not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null,
    updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Table 2: administrators
create table public.administrators (
    id uuid default uuid_generate_v4() primary key,
    user_id uuid references auth.users(id),
    full_name text not null,
    email text unique not null,
    role text not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Table 3: ministries
create table public.ministries (
    id uuid default uuid_generate_v4() primary key,
    ministry_name text unique not null,
    description text,
    ministry_leader text,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Table 4: church_events
create table public.church_events (
    id uuid default uuid_generate_v4() primary key,
    event_name text not null,
    event_description text,
    event_date date not null,
    event_location text not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Table 5: audit_logs
create table public.audit_logs (
    id uuid default uuid_generate_v4() primary key,
    administrator_id uuid references public.administrators(id),
    action text not null,
    record_type text not null,
    record_id uuid,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

2. Row Level Security (RLS) Policies

Secure your tables so public users can insert registrations, but only authenticated administrators can view or edit records:

alter table public.members enable row level security;

-- Allow public anonymous visitors to submit membership registration
create policy "Allow public membership registration" on public.members
    for insert with check (true);

-- Allow authenticated administrators full access to member records
create policy "Allow administrators full access to members" on public.members
    for all using (auth.role() = 'authenticated');
